-- DAR-LTCMS canonical realistic agricultural demo dataset
-- Updated for the A.O. No. 4, s. 2021 workflow and the current Approved/Denied
-- final-decision model. Release is tracked separately and never changes ownership.
--
-- Map geometry notes:
-- * Parcels are illustrative demo polygons only, not cadastral/legal boundaries.
-- * The demo parcels are distributed across four rural agricultural clusters in
--   San Jose, Valencia, and Dauin rather than the dense Dumaguete street grid.
-- * Parcel shapes are irregular, separated, and intentionally offset from barangay
--   centers/visible road corridors to read as agricultural holdings on the basemap.
-- * All seeded Parcel records are marked Private Agricultural Land.
-- * Final Approved/Denied demo applications leave the active landholding with the
--   transferor. No ownership/registry mutation is simulated.
--
-- Safe rerun rule: only records using the current NOR-AGRI / NOR-DEMO identifiers,
-- previous DGT-AGRI / DGT-DEMO demo identifiers, and the reserved 09996 demo
-- contact range are removed.

-- Capture only known DAR-LTCMS demo rows before any linked rows are removed.
-- This deliberately avoids blanket deletion of non-demo DGT/NOR application codes.
CREATE TEMP TABLE dar_demo_application_ids (id bigint PRIMARY KEY) ON COMMIT DROP;
INSERT INTO dar_demo_application_ids (id)
SELECT id
FROM land_transfer_applications
WHERE application_code LIKE '2026-DGT-DEMO-%'
   OR application_code LIKE '2026-NOR-DEMO-%'
UNION
SELECT DISTINCT ap.land_transfer_application_id
FROM application_parcels ap
JOIN parcels p ON p.id = ap.parcel_id
WHERE p.parcel_code LIKE 'DGT-AGRI-%'
   OR p.parcel_code LIKE 'NOR-AGRI-%';

CREATE TEMP TABLE dar_demo_landowner_ids (id bigint PRIMARY KEY) ON COMMIT DROP;
INSERT INTO dar_demo_landowner_ids (id)
SELECT DISTINCT lh.landowner_id
FROM landholdings lh
JOIN parcels p ON p.id = lh.parcel_id
WHERE p.parcel_code LIKE 'DGT-AGRI-%'
   OR p.parcel_code LIKE 'NOR-AGRI-%'
UNION
SELECT id
FROM landowners
WHERE contact_number LIKE '09996%'
   OR contact_number LIKE '09997%'
   OR contact_number LIKE '09999%';

DO $$
BEGIN
    IF to_regclass('public.application_clearances') IS NOT NULL THEN
        DELETE FROM application_clearances
        WHERE land_transfer_application_id IN (SELECT id FROM dar_demo_application_ids);
    END IF;

    IF to_regclass('public.application_parcels') IS NOT NULL THEN
        DELETE FROM application_parcels
        WHERE land_transfer_application_id IN (SELECT id FROM dar_demo_application_ids);
    END IF;
END $$;

DELETE FROM land_transfer_applications
WHERE id IN (SELECT id FROM dar_demo_application_ids);

DELETE FROM legacy_records
WHERE (parcel_code LIKE 'DGT-AGRI-%' OR parcel_code LIKE 'NOR-AGRI-%')
   OR source_book IN (
        'Dumaguete Mock Agri Map 2026',
        'Dumaguete Agri Map Cluster 2026',
        'Dumaguete Realistic Agri Parcels 2026',
        'DAR-LTCMS Provincial Agricultural Demo 2026'
   );

DELETE FROM source_record_packages
WHERE (parcel_code LIKE 'DGT-AGRI-%' OR parcel_code LIKE 'NOR-AGRI-%')
   OR source_book IN (
        'Dumaguete Mock Agri Map 2026',
        'Dumaguete Agri Map Cluster 2026',
        'Dumaguete Realistic Agri Parcels 2026',
        'DAR-LTCMS Provincial Agricultural Demo 2026'
   );

DELETE FROM landholdings
WHERE parcel_id IN (
    SELECT id FROM parcels
    WHERE parcel_code LIKE 'DGT-AGRI-%' OR parcel_code LIKE 'NOR-AGRI-%'
);

DELETE FROM parcels
WHERE parcel_code LIKE 'DGT-AGRI-%' OR parcel_code LIKE 'NOR-AGRI-%';

            CASE WHEN v_has_csw THEN 'Demo Completed Staff Work entry for workflow presentation.' ELSE NULL END,
            CASE WHEN r.release_status IN ('ready_for_release','released') THEN CURRENT_TIMESTAMP - INTERVAL '1 day' ELSE NULL END,
            CASE WHEN r.release_status = 'released' THEN CURRENT_TIMESTAMP - INTERVAL '6 hours' ELSE NULL END,
            CASE WHEN r.release_status = 'released' THEN v_staff_id ELSE NULL END,
            CASE WHEN r.release_status = 'released' THEN r.transferee ELSE NULL END,
            CASE WHEN r.release_status = 'released' THEN 'LOG-DEMO-' || lpad(r.seq::text, 4, '0') ELSE NULL END,
            CASE WHEN r.release_status = 'released' THEN 'received' ELSE 'not_recorded' END,
            CASE WHEN v_is_final THEN v_staff_id ELSE NULL END,
            CASE WHEN v_is_final THEN CURRENT_TIMESTAMP - INTERVAL '2 days' ELSE NULL END,
            CASE
                WHEN r.status = 'approved' THEN 'PARPO II final Approved demo decision.'
                WHEN r.status = 'denied' THEN 'PARPO II final Denied demo decision.'
                ELSE NULL
            END,
            CASE WHEN v_is_final THEN 'Final demo decision is immutable; no ownership or registry mutation was performed.' ELSE NULL END,
            CASE WHEN v_is_final THEN CURRENT_TIMESTAMP - INTERVAL '2 days' ELSE NULL END,
            CASE WHEN v_is_final THEN jsonb_build_object(
                'demo_seed', true,
                'ownership_transfer_performed', false,
                'registry_mutation_performed', false,
                'scope_note', 'Administrative clearance decision only.'
            ) ELSE NULL END,
            900 + r.seq,
            CASE WHEN r.release_status = 'released' THEN CURRENT_DATE ELSE NULL END,
            now(), now()
        ) RETURNING id INTO v_application_id;

        INSERT INTO application_parcels (
            land_transfer_application_id, parcel_id, area_hectares,
            parcel_code, title_no, tax_decl_no, lot_number,
            survey_plan_number, title_type, rod_office, area_square_meters,
            created_at, updated_at
        ) VALUES (
            v_application_id, v_parcel_id, r.area,
            v_parcel_code, v_title_no, v_tax_decl_no, v_lot_no,
            v_survey_no, r.title_type, 'Negros Oriental Province',
            round((r.area * 10000)::numeric, 2), now(), now()
        ) RETURNING id INTO v_application_parcel_id;

        IF v_is_final THEN
            INSERT INTO application_clearances (
                land_transfer_application_id, clearance_number, decision_status,
                application_code, transferor_name, transferee_name,
                municipality, barangay, total_area_hectares, parcel_snapshot,
                review_officer_name, reviewed_at, generated_by, generated_at,
                created_at, updated_at
            ) VALUES (
                v_application_id,
                '1803-2026-' || lpad((9000 + r.seq)::text, 4, '0') || ' (' || (900 + r.seq)::text || ')',
                r.status,
                v_application_code, r.transferor, r.transferee,
                r.municipality, r.barangay, r.area,
                jsonb_build_array(jsonb_build_object(
                    'parcel_id', v_parcel_id,
                    'parcel_code', v_parcel_code,
                    'title_no', v_title_no,
                    'tax_decl_no', v_tax_decl_no,
                    'lot_number', v_lot_no,
                    'survey_plan_number', v_survey_no,
                    'area_hectares', r.area
                )),
                'DAR-LTCMS Demo Review',
                CURRENT_TIMESTAMP - INTERVAL '2 days',
                v_staff_id,
                CURRENT_TIMESTAMP - INTERVAL '2 days',
                now(), now()
            );
        END IF;
    END LOOP;
END $$;
